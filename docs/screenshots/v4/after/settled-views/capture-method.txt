const output='docs/screenshots/v4/after/settled-views',baseURL='http://127.0.0.1:4001';
const archive=JSON.parse(await readFile('docs/screenshots/v4/after/manifest.json','utf8'));
const sha=data=>createHash('sha256').update(data).digest('hex');
for(const[file,expected]of Object.entries(archive.sourceHashes))if(sha(await readFile(file))!==expected)throw new Error(`Changed source: ${file}`);
await mkdir(output,{recursive:true});const records=[],files=[],issues=[];
const browser=await chromium.launch({executablePath:'/usr/bin/chromium'});
async function scrollInstant(page,top){for(let attempt=0;attempt<10;attempt++){await page.evaluate(async top=>{window.scrollTo({top,left:0,behavior:'instant'});await new Promise(resolve=>requestAnimationFrame(()=>requestAnimationFrame(resolve)));},top);await page.waitForTimeout(50);const actual=await page.evaluate(()=>({y:scrollY,max:Math.max(0,document.documentElement.scrollHeight-innerHeight)}));const expected=Math.min(Math.max(top,0),actual.max);if(Math.abs(actual.y-expected)<=0.5)return actual.y;}throw new Error(`Scroll did not settle at ${top}`);}
async function capture(page,file){await page.screenshot({path:`${output}/${file}`,fullPage:false,type:'jpeg',quality:82,animations:'disabled'});files.push({file,sha256:sha(await readFile(`${output}/${file}`))});}
try{for(const width of archive.widths){const height=width<768?844:width<1280?1024:900;const context=await browser.newContext({viewport:{width,height},deviceScaleFactor:1});const page=await page.evaluate(()=>document.fonts.ready);await page.waitForTimeout(150);await page.evaluate(()=>Promise.all([...document.images].filter(i=>i.complete||i.loading!=='lazy').map(i=>i.decode().catch(()=>null))));const scrollY=await scrollInstant(page,0);await capture(page,`${name}-${width}-fold.jpg`);records.push({name,route,width,height,status:response.status(),finalPath:new URL(page.url()).pathname,scrollY,overflow:await page.evaluate(()=>document.documentElement.scrollWidth-innerWidth)});if(name==='home'){const sections=await page.locator('main top=Math.max(0,section.y-86);const actual=await scrollInstant(page,top);const file=`home-${width}-section-${index+1}.jpg`;await readFile('/tmp/capture-v4-settled.txt');await writeFile(`${output}/capture-method.txt`,code);files.push({file:'capture-method.txt',sha256:sha(code)});await writeFile(`${output}/manifest.json`,JSON.stringify({capturedAt:new Date().toISOString(),baseURL,productionBuildId:archive.productionBuildId,sourceState:archive.sourceState,sourceCommit:archive.sourceCommit,sourceHashes:archive.sourceHashes,browser:await browser.version(),methodology:'Correction for viewport-only artifacts: CSS smooth scrolling caused some earlier fold/transition images to capture mid-scroll. Earlier files preserved. These authoritative viewport captures set behavior:instant, await two animation frames, assert actual scrollY against the bounded target, then capture. Full-page evidence, route/action/fold coordinates and interaction results remain in original manifest. Product sources verified unchanged before and after this supplementary run.',records,issues,files},null,2)+'\n');}finally{await browser.close();}
